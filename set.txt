 41 /* Read n elememts of a subset and store as a bit string */
 42 void readset(bit a[],int n)
 43 {
 44     int i,k,x;
 45     if(n<0 || n>SIZE){
 46         printf("Invalid size; must be between 0 and %d.\n",SIZE);
 47         return;
 48     }

 52         for(k=0;k<SIZE;k++){
 53             if(uset[k]==x){a[k].x=1;break;}
 54             }

 56                 printf("%d is not in the universal set,ignored.\n",x);
 57             }
 58         }
