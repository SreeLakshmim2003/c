 1 #include<stdio.h>
  2 #include<stdlib.h>
  3 struct Node{
  4     int data;
  5     struct Node*left,*right;
  6 };
  7 struct Node*insert(struct Node*root,int x){
  8     if(root==NULL){
  9         root=malloc(sizeof(struct Node));
 10         root->data=x;
 11         root->left=root->right=NULL;
 12     }
 13     else if(x<root->data)root->left=insert(root->left,x);
 14     else
 15         root->right=insert(root->right,x);
 16     return root;
 17 }
 18 void inorder(struct Node*root){
 19     if(root!=NULL){
 20         inorder(root->left);
 21         printf("%d",root->data);
 22         inorder(root->right);
 23     }
 24 }
 25 void preorder(struct Node*root)

 28         printf("%d",root->data);

 31     }
 32 }
 33 void postorder(struct Node*root){
 34     if(root!=NULL){
 35         postorder(root->left);

 38     }
 39 }
